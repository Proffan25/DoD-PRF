province_event = {
	id = 78700943
	title = "Horn of Africa Core Debugging"
	desc = "If there is an awkward overlapping culture core, this event is intended to clean it up"
	
	trigger = {
		owner = {
			OR = {
				tag = ETH
				tag = ADA
				tag = ERI
				tag = SOM
			}
			NOT = {
				war = yes
			}
		}
		OR = {
			AND = {
				is_core = ETH
				is_core = ADA
			}
			AND = {
				is_core = ETH
				is_core = ERI
			}
			AND = {
				is_core = ETH
				is_core = SOM
			}
			AND = {
				is_core = ADA
				is_core = ERI
			}
			AND = {
				is_core = ADA
				is_core = SOM
			}
			AND = {
				is_core = SOM
				is_core = ERI
			}
		}
	}
	
	mean_time_to_happen = {
		days = 1
	}
	
	option = {
		name = "Fix me up!"
		remove_core = ETH
		remove_core = ADA
		remove_core = ERI
		remove_core = SOM
		add_core = THIS
	}
}
