#Starting 1836 Literacy Fix for minority POPs in Colombia
country_event = {
	id = 123412346
	title = "Education Fix"
	desc = "for minorities in Colombia"
	picture = "Administration"
	
	
	trigger = {
		year = 1836
		tag = CLM
	} 

	fire_only_once = yes
	
	mean_time_to_happen = {
		days = 1
	}
	
	option = {
		name = "Learn 2 read dud"
		any_pop = {
			limit = {
				or = {				
					has_pop_culture = central_american
					has_pop_culture = granadan
					has_pop_culture = lusitanian
					has_pop_culture = quechua
				}
			}
			literacy = 0.12 #20 percent increase
		}
		any_owned = {
			any_pop = {
				limit = { 
					is_primary_culture = no
					is_accepted_culture = no
					type = bureaucrats
				}
				pop_type = artisans
			}				
		}
	}
}
